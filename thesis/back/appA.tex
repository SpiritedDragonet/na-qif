\section{12维发射体的块结构与基序置换}[Block Structure of 12D Emitter and Basis Permutation]

正文采用 atom-major 编码，索引写作
\begin{equation}
p=3i_a+i_c,\qquad i_a\in\{0,1,2,3\},\ i_c\in\{0,1,2\},
\label{eq:app-index-atom-major}
\end{equation}
其中 $i_a$ 对应 $(|0\rangle,|1\rangle,|e\rangle,|u\rangle)$，$i_c$ 对应 $(|{\rm vac}\rangle,|H\rangle,|V\rangle)$。若改用 cavity-major 编码，
\begin{equation}
q=4i_c+i_a.
\label{eq:app-index-cavity-major}
\end{equation}
定义置换矩阵 $P\in\mathbb{C}^{12\times 12}$ 为
\begin{equation}
P_{q,p}=
\begin{cases}
1,& q=4i_c+i_a,\ p=3i_a+i_c,\\
0,& \text{其他}.
\end{cases}
\label{eq:app-perm}
\end{equation}
则两种编码下的发射体哈密顿量满足
\begin{equation}
\tilde H_{\rm em}=P\,H_{\rm em}\,P^\dagger.
\label{eq:app-heqv}
\end{equation}

把 \eqref{eq:app-heqv} 展开为 $3\times 3$ 的 $4\times 4$ 块后得到
\begin{equation}
\tilde H_{\mathrm{em}}(t)=
\begin{bmatrix}
H_{\mathrm{atom}}(t) & g\,\sigma_H^\dagger & g\,\sigma_V^\dagger\\
g\,\sigma_H & H_{\mathrm{atom}}(t)+\delta_{c,H}I_4 & 0\\
g\,\sigma_V & 0 & H_{\mathrm{atom}}(t)+\delta_{c,V}I_4
\end{bmatrix}.
\label{eq:app-hem-block}
\end{equation}
式 \eqref{eq:app-hem-block} 显式区分了三类项：原子内部演化（对角）、真空-单光子腔模交换（非对角）以及偏振失谐修正（对角偏移）。

\section{从 36 维子门到 60 维站点门的索引嵌入}[Index Embedding from 36D Sub-Gate to 60D Site Gate]

发射生成元在 $(12\times 3)$ 子空间写为
\begin{equation}
G_{36}
=
\sqrt{\Delta t}\sum_{\mu\in\{H,V\}}
\left(L_{{\rm out},\mu}\otimes b_\mu^\dagger-L_{{\rm out},\mu}^\dagger\otimes b_\mu\right)
-i\Delta t\,H_{\rm em}\otimes I_3.
\label{eq:app-g36}
\end{equation}
由于每项均为反厄米，$G_{36}^\dagger=-G_{36}$，故
\begin{equation}
U_{36}=e^{G_{36}}
\end{equation}
严格幺正。

设完整两站点基序为 $(12\mathrm{D}\otimes 5\mathrm{D})$，索引
\begin{equation}
r=5i_e+i_b,\qquad c=5j_e+j_b,
\end{equation}
其中 $i_e,j_e\in\{0,\dots,11\}$，$i_b,j_b\in\{0,\dots,4\}$。定义 60 维门 $U_{60}$：
\begin{equation}
U_{60}[r,c]=
\begin{cases}
U_{36}[3i_e+i_b,\ 3j_e+j_b], & i_b<3,\ j_b<3,\\
\delta_{rc}, & \text{否则}.
\end{cases}
\label{eq:app-u60-piecewise}
\end{equation}
这正是“780 子块更新、1517 子块恒等”的精确索引表达。式 \eqref{eq:app-u60-piecewise} 也直接给出实现检查条件：若输入、输出索引任一落在 $i_b\ge 3$，矩阵元必须等于单位阵对应元。

\section{滤波记忆步进门的离散推导与幺正性}[Discrete Derivation and Unitarity of Memory-Filter Step Gate]

从单极点滤波方程
\begin{equation}
\dot a_f(t)= -\left(\frac{\kappa_f}{2}+i\delta\right)a_f(t)+\sqrt{\kappa_f}\,b_{\rm in}(t)
\label{eq:app-filter-ode}
\end{equation}
在步长 $\Delta t$ 上作一阶精确离散，得到
\begin{equation}
a_{n+1}=r\,a_n+t\,b_n,\qquad
\tilde b_n=r^*\,b_n-t\,a_n,
\label{eq:app-filter-discrete}
\end{equation}
其中
\begin{equation}
r=\exp\!\left[-(\pi\Delta\nu+i2\pi\delta)\Delta t\right],\qquad
t=\sqrt{1-|r|^2}.
\label{eq:app-rt}
\end{equation}

对应的二维子块矩阵
\begin{equation}
B=
\begin{bmatrix}
r&-t\\
t&r^*
\end{bmatrix}
\label{eq:app-B}
\end{equation}
满足
\begin{equation}
B^\dagger B=
\begin{bmatrix}
|r|^2+t^2&0\\
0&|r|^2+t^2
\end{bmatrix}
=I_2.
\end{equation}
因此 $B$ 幺正，两个偏振子块直和后仍幺正，进而整个 $(5\times 3)=15$ 维步进门幺正。

\section{交换调度的不变量与终态布局}[Swap-Schedule Invariants and Final Layout]

发射阶段与记忆阶段都依赖“相邻交换 + 局部门”调度。其正确性的核心不变量为：
\begin{equation}
\mathcal{I}_1:\ \text{每轮门作用前，系统体右邻/左邻关系满足预设的相邻约束},
\end{equation}
\begin{equation}
\mathcal{I}_2:\ \text{每轮结束后，仅系统体索引改变，bin 相对时序不改变}.
\end{equation}

以发射阶段为例，若记某轮前 atomA、atomB 站点为 $p_A,p_B$，则门作用后采用固定交换模板
\begin{equation}
\mathsf{S}_{p_A-1}\mathsf{S}_{p_A-2},\qquad
\mathsf{S}_{p_B-1}\mathsf{S}_{p_B-2}
\end{equation}
（末轮对 atomA 保留一次交换），可归纳证明最终前缀必为 $(\mathrm{atomA,atomB})$。记忆阶段同理：每轮先把 $\mathrm{memA},\mathrm{memB}$ 交换到目标 bin 右侧，再施加步进门，最后收拢到前缀 $(\mathrm{atomA,atomB,memA,memB})$。

该归纳证明的关键不在具体交换次数，而在“目标相邻关系 + 单调移动方向”两条约束同时成立。只要两者保持，链长与窗口变化不会改变终态布局结论。

\section{计时闭合关系与主导项判据}[Timing Closure and Dominant-Term Criterion]

为与运行剖面一致，定义
\begin{equation}
T_w=T_{\rm phy}+T_{\rm det,b}+T_{\rm det,m}+T_{\rm aux}+T_{\rm res},
\label{eq:app-time-closure}
\end{equation}
其中
\begin{equation}
T_{\rm phy}=T_{\rm em}+T_{\rm qm}+T_{\rm fib}+T_{\rm dep},
\end{equation}
\begin{equation}
T_{\rm det,b}=T_{\rm b,eff}+T_{\rm b,enum}+T_{\rm b,samp},\qquad
T_{\rm det,m}=T_{\rm m,eff}+T_{\rm m,enum}+T_{\rm m,samp},
\end{equation}
\begin{equation}
T_{\rm aux}=T_{\rm chk}+T_{\rm hk,p}+T_{\rm hk,s}+T_{\rm par}+T_{\rm met}+T_{\rm io,s}+T_{\rm io,d}+T_{\rm post}.
\end{equation}

据此可定义无量纲主导比
\begin{equation}
\rho_{\rm enum}=\frac{T_{\rm b,enum}+T_{\rm m,enum}}{T_w},\qquad
\rho_{\rm phy}=\frac{T_{\rm phy}}{T_w}.
\end{equation}
当 $\rho_{\rm enum}\gg \rho_{\rm phy}$ 时，优化优先级应落在 effect 枚举与收缩；反之才应优先优化态端动力学。该判据比单看 $T_w$ 更能避免误判瓶颈位置。
